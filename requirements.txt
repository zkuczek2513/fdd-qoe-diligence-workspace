# Upper bounds are pinned to the newest release verified against the full test
# suite (see README "Testing"). The app is exercised at both ends of every
# range — Streamlit 1.37 and 1.60, pandas 2.2 and 3.0, Plotly 5.24 and 6.9 —
# so a deployment reproduces what was tested instead of picking up an
# unverified release at build time.

pandas>=2.0,<4

# Module 2 — SEC EDGAR ingestion.
# Deliberately no lxml: it is a compiled extension whose wheels lag new Python
# releases (no lxml 5.x publishes a cp314 wheel, so a capped pin forces a source
# build that fails on hosts without libxml2 headers). edgar_client uses lxml when
# it happens to be present and falls back to a regex HTML stripper otherwise —
# the two produce byte-identical output on the filings tested, so the dependency
# bought nothing and cost a deployment.
aiohttp>=3.9,<5

# Module 7 — report export (pure Python, no build step)
